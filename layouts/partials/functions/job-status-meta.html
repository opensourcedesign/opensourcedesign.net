{{/* Normalized job status for templates, SEO, and feeds. Canonical values: searching | filled | closed | expired Legacy solved/resolved/completed map to filled. */}} {{ $raw := lower (printf "%v" (.Params.status | default "")) }} {{ $status := $raw }} {{ if in (slice "solved" "resolved" "completed") $raw }}{{ $status = "filled" }}{{ end }} {{ $isOpen := eq $status "searching" }} {{/* A "searching" posting past its deadline or a year old counts as expired before job-expire's status flip lands (functions/job-lapsed.html). */}} {{ $isExpired := or (eq $status "expired") (and $isOpen (partial "functions/job-lapsed.html" .)) }} {{ $noindex := or $isExpired (in (slice "filled" "closed" "expired") $status) }} {{ return (dict "raw" $raw "status" $status "open" $isOpen "expired" $isExpired "noindex" $noindex) }}
